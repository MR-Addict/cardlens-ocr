cmake_minimum_required(VERSION 3.20)
project(cardlens_ocr VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(CMAKE_GENERATOR MATCHES "Ninja")
  set_property(GLOBAL PROPERTY JOB_POOLS cardlens_compile=1 cardlens_link=1)
  set(CMAKE_JOB_POOL_COMPILE cardlens_compile)
  set(CMAKE_JOB_POOL_LINK cardlens_link)
endif()
if(MSVC)
  add_compile_options(/utf-8)
endif()
option(CARDLENS_TESTS "Build native tests" ON)
set(CARDLENS_DEPS "${CMAKE_SOURCE_DIR}/out/deps/host" CACHE PATH "Installed dependencies")
set(CARDLENS_COMMON "${CMAKE_SOURCE_DIR}/out/deps/common" CACHE PATH "Header and geometry dependencies")
find_package(ncnn REQUIRED CONFIG PATHS "${CARDLENS_DEPS}/lib/cmake/ncnn" NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH)
set(cv_modules core imgproc)
if(NOT EMSCRIPTEN AND NOT CMAKE_SYSTEM_NAME STREQUAL "OHOS")
  list(APPEND cv_modules imgcodecs)
endif()
find_package(OpenCV REQUIRED CONFIG COMPONENTS ${cv_modules} PATHS "${CARDLENS_DEPS}/lib/cmake/opencv4" NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH)
add_library(cardlens_core STATIC core/src/core.cpp core/src/fields.cpp core/src/ctc.cpp
  "${CARDLENS_COMMON}/clipper2/src/clipper.engine.cpp"
  "${CARDLENS_COMMON}/clipper2/src/clipper.offset.cpp"
  "${CARDLENS_COMMON}/clipper2/src/clipper.rectclip.cpp")
target_include_directories(cardlens_core PUBLIC core/include PRIVATE core/src "${CARDLENS_COMMON}/include" "${CARDLENS_COMMON}/clipper2/include" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(cardlens_core PUBLIC ncnn ${OpenCV_LIBS})
if(EMSCRIPTEN)
  add_executable(cardlens_wasm sdk/c/c_api.cpp)
  target_link_libraries(cardlens_wasm PRIVATE cardlens_core)
  target_compile_options(cardlens_core PRIVATE -fexceptions)
  target_compile_options(cardlens_wasm PRIVATE -fexceptions)
  target_link_options(cardlens_wasm PRIVATE
    -fexceptions -Wl,--threads=1 -sDISABLE_EXCEPTION_CATCHING=0 -sMODULARIZE=1 -sEXPORT_ES6=1
    -sENVIRONMENT=web,worker,node -sALLOW_MEMORY_GROWTH=1 -sINITIAL_MEMORY=67108864
    -sMAXIMUM_MEMORY=2147483648 -sSTACK_SIZE=5242880 -sFILESYSTEM=1
    "-sEXPORTED_FUNCTIONS=['_cl_initialize','_cl_recognize','_cl_release','_cl_free','_malloc','_free']"
    "-sEXPORTED_RUNTIME_METHODS=['FS','HEAPU8','UTF8ToString','stringToUTF8','lengthBytesUTF8','getValue','setValue']")
  set_target_properties(cardlens_wasm PROPERTIES OUTPUT_NAME cardlens SUFFIX .mjs
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/out/wasm")
elseif(CMAKE_SYSTEM_NAME STREQUAL "OHOS")
  add_library(cardlens_ocr SHARED sdk/harmonyos/native/napi_module.cpp)
  target_link_libraries(cardlens_ocr PRIVATE cardlens_core libace_napi.z.so)
  target_link_options(cardlens_ocr PRIVATE -Wl,--no-undefined)
else()
  target_sources(cardlens_core PRIVATE core/src/trace.cpp)
  target_compile_definitions(cardlens_core PUBLIC CARDLENS_OCR_DIAGNOSTICS)
  add_library(cardlens SHARED sdk/c/c_api.cpp)
  target_compile_definitions(cardlens PRIVATE CARDLENS_EXPORTS)
  target_link_libraries(cardlens PRIVATE cardlens_core)
  add_executable(cardlens-ocr core/src/cli.cpp)
  target_include_directories(cardlens-ocr PRIVATE core/src "${CARDLENS_COMMON}/include" ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(cardlens-ocr PRIVATE cardlens_core)
  if(CARDLENS_TESTS)
    enable_testing()
    add_executable(ctc-test tests/ctc_test.cpp)
    add_executable(core-test tests/core_test.cpp)
    foreach(t ctc-test core-test)
      target_include_directories(${t} PRIVATE core/src "${CARDLENS_COMMON}/include" ${OpenCV_INCLUDE_DIRS})
      target_link_libraries(${t} PRIVATE cardlens_core)
    endforeach()
    add_test(NAME ctc COMMAND ctc-test)
    add_test(NAME core COMMAND core-test "${CMAKE_SOURCE_DIR}/models" "${CMAKE_SOURCE_DIR}/out/testdata/id-front.jpg")
  endif()
  install(TARGETS cardlens cardlens_core cardlens-ocr)
  install(DIRECTORY core/include/ DESTINATION include)
endif()
